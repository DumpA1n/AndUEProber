cmake_minimum_required(VERSION 3.22.1)
project(AndUEProberAgentConsumer LANGUAGES C)
if(NOT ANDROID)
    message(FATAL_ERROR "The public Agent DSO requires Android")
endif()
if(ANDUEPROBER_SOURCE)
    set(ANDUEPROBER_BUILD_AGENT ON)
    set(ANDUEPROBER_BUILD_TESTS OFF)
    add_subdirectory(${ANDUEPROBER_SOURCE} andueprober)
else()
    find_package(AndUEProber 0.2 CONFIG REQUIRED COMPONENTS Agent)
    if(TARGET AndUEProber::Core OR TARGET AndUEProber::Inspector OR TARGET AndUEProber::ProcessMemory)
        message(FATAL_ERROR "Agent-only consumption must not import static implementation components")
    endif()
endif()
add_executable(agent_consumer main.c)
target_link_libraries(agent_consumer PRIVATE AndUEProber::Agent)
