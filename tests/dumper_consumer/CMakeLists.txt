cmake_minimum_required(VERSION 3.22.1)
project(AndUEProberDumperConsumer LANGUAGES CXX)
if(ANDUEPROBER_SOURCE)
    set(ANDUEPROBER_BUILD_AGENT OFF)
    set(ANDUEPROBER_BUILD_TESTS OFF)
    set(ANDUEPROBER_BUILD_INSPECTOR OFF)
    set(ANDUEPROBER_WITH_PROCESS_MEMORY OFF)
    set(ANDUEPROBER_BUILD_DUMPER_ADAPTER ON)
    add_subdirectory("${ANDUEPROBER_SOURCE}" andueprober)
else()
    find_package(AndUEProber 0.2 CONFIG REQUIRED COMPONENTS DumperAdapter)
    foreach(component Agent Inspector ProcessMemory)
        if(TARGET AndUEProber::${component})
            message(FATAL_ERROR "DumperAdapter must not import ${component}")
        endif()
    endforeach()
endif()
add_library(owned_dumper SHARED owned.cpp)
target_link_libraries(owned_dumper PRIVATE AndUEProber::DumperAdapter)
add_executable(dumper_consumer main.cpp)
target_compile_features(dumper_consumer PRIVATE cxx_std_20)
target_link_libraries(dumper_consumer PRIVATE ${CMAKE_DL_LIBS})
if(ANDUEPROBER_SANITIZERS)
    target_compile_options(dumper_consumer PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(dumper_consumer PRIVATE -fsanitize=address,undefined)
endif()
foreach(header DumperAdapter.hpp Reflection.hpp)
    string(REPLACE "." "_" name "${header}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${name}.cpp" "#include <andueprober/${header}>\n")
    add_library(header_${name} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/${name}.cpp")
    target_link_libraries(header_${name} PRIVATE AndUEProber::Core)
endforeach()
enable_testing()
add_test(NAME andueprober.dumper_consumer COMMAND dumper_consumer $<TARGET_FILE:owned_dumper>
    "${CMAKE_CURRENT_BINARY_DIR}/generated/Reflection.hpp" "${CMAKE_CURRENT_BINARY_DIR}/owned-export")
add_test(NAME andueprober.dumper_generated_compile COMMAND ${CMAKE_COMMAND}
    "-DHEADER_DIR=${CMAKE_CURRENT_BINARY_DIR}/generated" "-DSMOKE_SOURCE=${CMAKE_CURRENT_SOURCE_DIR}/smoke"
    "-DSMOKE_BUILD=${CMAKE_CURRENT_BINARY_DIR}/smoke" "-DCXX=${CMAKE_CXX_COMPILER}"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/compile.cmake")
set_tests_properties(andueprober.dumper_consumer PROPERTIES FIXTURES_SETUP dumper_header)
set_tests_properties(andueprober.dumper_generated_compile PROPERTIES FIXTURES_REQUIRED dumper_header)
