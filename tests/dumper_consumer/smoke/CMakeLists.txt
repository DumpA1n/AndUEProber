cmake_minimum_required(VERSION 3.22.1)
project(GeneratedTypedDumperSmoke LANGUAGES CXX)
if(NOT EXISTS "${HEADER_DIR}/Reflection.hpp")
    message(FATAL_ERROR "HEADER_DIR must contain the actual generated Reflection.hpp")
endif()
add_executable(dumper_smoke main.cpp)
target_compile_features(dumper_smoke PRIVATE cxx_std_20)
target_include_directories(dumper_smoke PRIVATE "${HEADER_DIR}")
target_compile_options(dumper_smoke PRIVATE -Wall -Wextra -Werror -pedantic)
