cmake_minimum_required(VERSION 3.22.1)
project(AndUEProberComponentConsumer LANGUAGES CXX)

option(COMPONENT_WITH_INSPECTOR "Link the public Inspector into the caller DSO" OFF)
option(COMPONENT_WITH_MEMORY "Link the public ProcessMemory into the caller DSO" OFF)
option(COMPONENT_WITH_DUMPER "Link the public frozen reflection adapter into the caller DSO" OFF)
if(ANDUEPROBER_SOURCE)
    set(ANDUEPROBER_BUILD_AGENT OFF)
    set(ANDUEPROBER_BUILD_TESTS OFF)
    set(ANDUEPROBER_BUILD_INSPECTOR ${COMPONENT_WITH_INSPECTOR})
    set(ANDUEPROBER_WITH_PROCESS_MEMORY ${COMPONENT_WITH_MEMORY})
    set(ANDUEPROBER_BUILD_DUMPER_ADAPTER ${COMPONENT_WITH_DUMPER})
    add_subdirectory(${ANDUEPROBER_SOURCE} andueprober)
else()
    set(components Core Probe)
    if(COMPONENT_WITH_INSPECTOR)
        list(APPEND components Inspector)
    endif()
    if(COMPONENT_WITH_MEMORY)
        list(APPEND components ProcessMemory)
    endif()
    if(COMPONENT_WITH_DUMPER)
        list(APPEND components DumperAdapter)
    endif()
    find_package(AndUEProber 0.2 CONFIG REQUIRED COMPONENTS ${components})
    if(NOT COMPONENT_WITH_MEMORY AND TARGET AndUEProber::ProcessMemory)
        message(FATAL_ERROR "Core/Probe consumption must not import ProcessMemory")
    endif()
endif()

add_library(owned_component SHARED component.cpp)
target_link_libraries(owned_component PRIVATE AndUEProber::Probe)
if(COMPONENT_WITH_INSPECTOR)
    target_compile_definitions(owned_component PRIVATE COMPONENT_WITH_INSPECTOR=1)
    target_link_libraries(owned_component PRIVATE AndUEProber::Inspector AndUEProberImGui::Binding)
endif()
if(COMPONENT_WITH_MEMORY)
    target_compile_definitions(owned_component PRIVATE COMPONENT_WITH_MEMORY=1)
    target_link_libraries(owned_component PRIVATE AndUEProber::ProcessMemory)
endif()
if(COMPONENT_WITH_DUMPER)
    target_compile_definitions(owned_component PRIVATE COMPONENT_WITH_DUMPER=1)
    target_link_libraries(owned_component PRIVATE AndUEProber::DumperAdapter)
endif()
foreach(header Core.hpp Evidence.hpp Export.hpp Reflection.hpp DumperAdapter.hpp Probe.hpp Inspector.hpp ProcessMemory.hpp)
    string(REPLACE "." "_" name "${header}")
    file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/header_${name}.cpp "#include <andueprober/${header}>\n")
    add_library(component_header_${name} OBJECT ${CMAKE_CURRENT_BINARY_DIR}/header_${name}.cpp)
    target_link_libraries(component_header_${name} PRIVATE AndUEProber::Core)
endforeach()
add_executable(component_consumer main.cpp)
target_compile_features(component_consumer PRIVATE cxx_std_20)
target_link_libraries(component_consumer PRIVATE ${CMAKE_DL_LIBS})
if(ANDUEPROBER_SANITIZERS)
    target_compile_options(component_consumer PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(component_consumer PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME andueprober.component_consumer
    COMMAND component_consumer $<TARGET_FILE:owned_component> ${CMAKE_CURRENT_BINARY_DIR}/owned-output)
