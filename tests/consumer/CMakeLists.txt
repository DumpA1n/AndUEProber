cmake_minimum_required(VERSION 3.22)
project(AndUEProberConsumer LANGUAGES CXX)
if(ANDUEPROBER_SOURCE)
    set(ANDUEPROBER_BUILD_AGENT OFF CACHE BOOL "" FORCE)
    set(ANDUEPROBER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    add_subdirectory(${ANDUEPROBER_SOURCE} andueprober)
else()
    find_package(AndUEProber 0.2 CONFIG REQUIRED COMPONENTS Core Probe)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE AndUEProber::Probe)
